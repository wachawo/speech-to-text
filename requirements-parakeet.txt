# Parakeet transcription backend, installed only when the image is built with
# --build-arg PARAKEET=true. Mirrors the `parakeet` extra in pyproject.toml.
#
# Unlike the diarizer, this one needs no git pin: parakeet_tdt is in released transformers,
# and AutoModelForTDT exists there. Kept as a floor rather than an exact pin so it can
# co-exist with the diarize extra, which pulls a newer transformers from a commit.
transformers>=5.17.0

# Reading the 16 kHz mono WAV the server hands over.
soundfile>=0.12.1

# ParakeetFeatureExtractor is declared @requires(backends=("torch", "librosa")) and builds its
# mel filterbank with librosa.filters.mel, the same trap as the diarizer's feature extractor.
# Not `transformers[audio]`: that extra also drags in an unconstrained torchaudio.
librosa>=0.11.0
